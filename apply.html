/** * 서울오라토리오 합창단 지원 신청서 — 접수 서버 (Google Apps Script) * * 하는 일 * 1) 지원서 PDF를 구글 드라이브 「서울오라토리오 지원신청서」 폴더에 저장 * 2) 이 스프레드시트의 「접수목록」 시트에 한 줄씩 기록 * 3) 담당자 메일로 PDF 첨부해 알림 발송 * 4) 지원자가 E-mail을 적었으면 사본 발송 */ // ▼ 지원서를 받을 담당자 메일 주소 (여러 명이면 쉼표로 구분) const NOTIFY_EMAIL = 'skysong7788@naver.com'; const FOLDER_NAME = '서울오라토리오 지원신청서'; const SHEET_NAME = '접수목록'; function doPost(e) { try { const d = JSON.parse(e.postData.contents); if (!d.name || !d.mobile || !d.pdf) throw new Error('필수 항목 누락'); // 1) PDF 저장 const blob = Utilities.newBlob(Utilities.base64Decode(d.pdf), 'application/pdf', d.fileName); const file = getFolder_().createFile(blob); // 2) 시트 기록 const sheet = getSheet_(); sheet.appendRow([ new Date(), d.name, d.nameHanja, d.nameEn, d.birth, d.age, d.married, d.zip, d.addr, d.tel, d.mobile, d.email, d.work, d.religion, d.route, d.career, file.getUrl() ]); // 3) 담당자 알림 MailApp.sendEmail({ to: NOTIFY_EMAIL, subject: `[합창단 지원] ${d.name} (${d.mobile})`, htmlBody: `

새 지원 신청서가 접수되었습니다.

성명 ${esc_(d.name)}
휴대폰 ${esc_(d.mobile)}
E-mail ${esc_(d.email || '-')}
입단 경로 ${esc_(d.route || '-')}

드라이브에서 PDF 열기

`, attachments: [blob] }); // 4) 지원자 사본 if (d.email) { MailApp.sendEmail({ to: d.email, subject: '[서울오라토리오] 합창단 지원 신청서가 접수되었습니다', htmlBody: `

${esc_(d.name)}님, 서울오라토리오 합창단에 지원해 주셔서 감사합니다.

문의: 02-587-9277 / 010-2683-6309

`, attachments: [blob], name: '서울오라토리오' }); } return json_({ ok: true }); } catch (err) { return json_({ ok: false, error: String(err) }); } } function getFolder_() { const it = DriveApp.getFoldersByName(FOLDER_NAME); return it.hasNext() ? it.next() : DriveApp.createFolder(FOLDER_NAME); } function getSheet_() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sh = ss.getSheetByName(SHEET_NAME); if (!sh) { sh = ss.insertSheet(SHEET_NAME); sh.appendRow(['접수일시','성명','한자','영문','생년월일','나이','결혼','우편번호','주소', '전화번호','휴대폰','E-mail','현재근무처','종교(교회명)','입단 경로', '연주경력과 현 활동사항','신청서 PDF']); sh.setFrozenRows(1); sh.getRange(1, 1, 1, 17).setFontWeight('bold'); } return sh; } function json_(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); } function esc_(s) { return String(s || '').replace(/[&<>"]/g, c => ({ '&': '&', '<': '<', '>': '>', '"': '"' }[c])); } // 배포 전 권한 승인용 — 편집기에서 한 번 실행하세요 function authorize() { getFolder_(); getSheet_(); MailApp.getRemainingDailyQuota(); }
